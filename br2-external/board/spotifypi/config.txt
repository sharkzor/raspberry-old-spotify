# SpotifyPi - Raspberry Pi 2 firmware configuration.
# This file lives on the FAT boot partition and can be edited from any PC.

start_file=start_cd.elf
fixup_file=fixup_cd.dat
kernel=zImage
gpu_mem=16
disable_overscan=1
disable_splash=1
boot_delay=0

# Onboard (PWM) audio off, I2S DAC on.
dtparam=audio=off
dtparam=i2s=on
# DAC overlay. Pick the one matching your board:
#   hifiberry-dac          PCM5102A boards (no I2C control)
#   hifiberry-dacplus      PCM5122 boards (DAC+, many clones such as NXEZ)
#   hifiberry-dacplus-std  PCM5122 clones without onboard clock oscillators
#   (others: iqaudio-dacplus, allo-boss-dac-pcm512x-audio, ...)
dtoverlay=hifiberry-dacplus
